// --- Dijkstra (safety-limited — throws instead of hanging on a bad edit) --
function shortestPath(fromId: number, toId: number): number[] | null {
  const adjacency = new Map<number, { to: number; weight: number }[]>();
  for (const node of NODES) adjacency.set(node.nodeId, []);
  for (const edge of EDGES) {
    adjacency.get(edge.a)!.push({ to: edge.b, weight: edge.weight });
    adjacency.get(edge.b)!.push({ to: edge.a, weight: edge.weight });
  }

  const dist = new Map<number, number>(NODES.map((n) => [n.nodeId, Infinity]));
  const prev = new Map<number, number | null>();
  const visited = new Set<number>();
  dist.set(fromId, 0);

  let safety = 0;
  while (visited.size < NODES.length) {
    if (++safety > 2000) {
      console.error('shortestPath: safety limit hit', { fromId, toId });
      throw new Error('shortestPath exceeded safety limit — see console');
    }
    let current: number | null = null;
    let currentDist = Infinity;
    for (const [nodeId, dd] of dist) {
      if (!visited.has(nodeId) && dd < currentDist) {
        current = nodeId;
        currentDist = dd;
      }
    }
    if (current === null || current === toId) break;
    visited.add(current);
    for (const { to, weight } of adjacency.get(current)!) {
      const candidate = currentDist + weight;
      if (candidate < (dist.get(to) ?? Infinity)) {
        dist.set(to, candidate);
        prev.set(to, current);
      }
    }
  }

  if (dist.get(toId) === Infinity) return null;

  const path: number[] = [];
  let step: number | null = toId;
  let safety2 = 0;
  while (step !== null) {
    if (++safety2 > 2000) throw new Error('path reconstruction exceeded safety limit');
    path.unshift(step);
    step = prev.get(step) ?? null;
  }
  return path;
}

function buildRoute(fromNodeId: number, toNodeId: number): RouteResponse | null {
  const path = shortestPath(fromNodeId, toNodeId);
  if (!path) return null;

  const points: RoutePoint[] = path.map((id) => {
    const node = NODE_BY_ID.get(id)!;
    return { nodeId: node.nodeId, name: node.name, type: node.type, floorId: node.floorId, floorName: node.floorName, x: node.x, y: node.y };
  });

  let totalDistance = 0;
  for (let i = 1; i < path.length; i++) {
    const edge = EDGES.find((e) => (e.a === path[i - 1] && e.b === path[i]) || (e.b === path[i - 1] && e.a === path[i]));
    totalDistance += edge?.weight ?? 0;
  }

  return {
    venueId: MOCK_VENUE.id,
    fromNodeId,
    toNodeId,
    totalDistance: Math.round(totalDistance * 100) / 100,
    estimatedTimeMinutes: Math.max(1, Math.ceil(totalDistance / 90)),
    points,
  };
}

// --- Session state ---------------------------------------------------------
interface MockSession { sessionId: string; currentPoint: StartPoint; destinationNodeId: number | null; status: 'ACTIVE' | 'COMPLETED'; }
const sessions = new Map<string, MockSession>();
let nextSessionNumber = 1;

@Injectable({ providedIn: 'root' })
export class VisitorApiService {
  resolveQrCode(code: string, sessionId?: string | null): Observable<QrResolveResponse> {
    const nodeId = QR_TO_NODE_ID[code] ?? QR_TO_NODE_ID['main-entrance'];
    const node = NODE_BY_ID.get(nodeId)!;
    const startPoint = toStartPoint(node);
    const existing = sessionId ? sessions.get(sessionId) : undefined;
    const session = existing ?? this.createSession(startPoint);
    session.currentPoint = startPoint;
    session.status = 'ACTIVE';
    return of({ sessionId: session.sessionId, hasActiveJourney: session.destinationNodeId !== null, venue: MOCK_VENUE, startPoint, activeJourney: this.toCurrentJourney(session) });
  }

  getDestinations(venueId: number): Observable<Destination[]> {
    return venueId === MOCK_VENUE.id ? of(DESTINATIONS) : throwError(() => new Error('Venue not found'));
  }

  calculateRoute(sessionId: string, destinationNodeId: number): Observable<RouteResponse> {
    const session = sessions.get(sessionId);
    if (!session) return throwError(() => new Error('Session not found'));
    const route = buildRoute(session.currentPoint.nodeId, destinationNodeId);
    if (!route) return throwError(() => new Error('No path found to this destination'));
    session.destinationNodeId = destinationNodeId;
    return of(route);
  }

  getCurrentJourney(sessionId: string): Observable<CurrentJourney> {
    const session = sessions.get(sessionId);
    const journey = session ? this.toCurrentJourney(session) : null;
    return journey ? of(journey) : throwError(() => new Error('Session not found'));
  }

  updateLocation(sessionId: string, nodeId: number): Observable<RouteResponse> {
    const session = sessions.get(sessionId);
    const node = NODE_BY_ID.get(nodeId);
    if (!session || !node || session.destinationNodeId === null) return throwError(() => new Error('No active journey to update'));
    session.currentPoint = toStartPoint(node);
    const route = buildRoute(nodeId, session.destinationNodeId);
    return route ? of(route) : throwError(() => new Error('No path found to this destination'));
  }

  completeJourney(sessionId: string): Observable<CompleteJourneyResponse> {
    const session = sessions.get(sessionId);
    if (!session) return throwError(() => new Error('Session not found'));
    session.status = 'COMPLETED';
    session.destinationNodeId = null;
    return of({ sessionId, status: 'COMPLETED', venueId: MOCK_VENUE.id, venueName: MOCK_VENUE.name, resumable: false });
  }

  getFloorPlanMeta(floorId: number): Observable<FloorPlanMeta> {
    const plan = FLOOR_PLANS.find((p) => p.floorId === floorId);
    return plan ? of(plan) : throwError(() => new Error('Floor plan not found'));
  }

  getFloorPlanImageUrl(floorId: number): string {
    return FLOOR_PLAN_IMAGE_PATHS[floorId] ?? FLOOR_PLAN_IMAGE_PATHS[1];
  }

  private createSession(currentPoint: StartPoint): MockSession {
    const session: MockSession = { sessionId: `mock-session-${String(nextSessionNumber++).padStart(4, '0')}`, currentPoint, destinationNodeId: null, status: 'ACTIVE' };
    sessions.set(session.sessionId, session);
    return session;
  }

  private toCurrentJourney(session: MockSession): CurrentJourney | null {
    if (session.destinationNodeId === null) return null;
    const destinationNode = NODE_BY_ID.get(session.destinationNodeId);
    if (!destinationNode) return null;
    return {
      sessionId: session.sessionId, status: session.status, venueId: MOCK_VENUE.id, venueName: MOCK_VENUE.name,
      currentPoint: session.currentPoint,
      destination: { nodeId: destinationNode.nodeId, name: destinationNode.name, floorId: destinationNode.floorId, floorName: destinationNode.floorName },
      resumable: session.status === 'ACTIVE',
    };
  }
}
