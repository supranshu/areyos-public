@Injectable({ providedIn: 'root' })
export class VisitorApiService {
  resolveQrCode(code: string): Observable<QrResolveResponse> {
    return of({ venue: MOCK_VENUE, startPoint: MOCK_START_POINT });
  }

  getDestinations(venueId: number): Observable<Destination[]> {
    return of(MOCK_DESTINATIONS);
  }

  calculateRoute(venueId: number, fromNodeId: number, toNodeId: number): Observable<RouteResponse> {
    const destination = MOCK_DESTINATIONS.find((item) => item.nodeId === toNodeId) ?? MOCK_DESTINATIONS[0];
    const points: RoutePoint[] = [
      { ...MOCK_START_POINT, type: 'JUNCTION' },
      {
        nodeId: 401,
        name: 'Welcome Hall Junction',
        type: 'JUNCTION',
        floorId: 1,
        floorName: 'Ground Floor',
        x: 180,
        y: 372,
      },
    ];

    if (destination.floorId === 1) {
      points.push({
        nodeId: 402,
        name: 'Central Concourse',
        type: 'JUNCTION',
        floorId: 1,
        floorName: 'Ground Floor',
        x: 620,
        y: 372,
      });
    } else {
      points.push({
        nodeId: 403,
        name: 'Lift and Stair Core',
        type: 'ELEVATOR',
        floorId: 1,
        floorName: 'Ground Floor',
        x: 725,
        y: 526,
      });
      points.push({
        nodeId: 404,
        name: 'Lift and Stair Core',
        type: 'ELEVATOR',
        floorId: 2,
        floorName: 'First Floor',
        x: 725,
        y: 526,
      });
      points.push({
        nodeId: 405,
        name: 'Upper Concourse',
        type: 'JUNCTION',
        floorId: 2,
        floorName: 'First Floor',
        x: 605,
        y: 350,
      });
    }

    points.push(destination);

    return of({
      venueId,
      fromNodeId,
      toNodeId: destination.nodeId,
      totalDistance: destination.floorId === MOCK_START_POINT.floorId ? 150 : 260,
      estimatedTimeMinutes: destination.floorId === MOCK_START_POINT.floorId ? 3 : 5,
      points,
    });
  }

  getFloorPlanMeta(floorId: number): Observable<FloorPlanMeta> {
    return of(MOCK_FLOOR_PLANS.find((plan) => plan.floorId === floorId) ?? MOCK_FLOOR_PLANS[0]);
  }

  /** Used directly as an <img [src]> — no need to route binary bytes through HttpClient. */
  getFloorPlanImageUrl(floorId: number): string {
    return MOCK_FLOOR_PLAN_IMAGE_URLS[floorId] ?? MOCK_FLOOR_PLAN_IMAGE_URLS[1];
  }
}
