interface MockSession {
  sessionId: string;
  currentPoint: StartPoint;
  destinationNodeId: number | null;
  status: 'ACTIVE' | 'COMPLETED';
}

const sessions = new Map<string, MockSession>();
let nextSessionId = 1;

@Injectable({ providedIn: 'root' })
export class VisitorApiService {
  resolveQrCode(code: string, sessionId?: string | null): Observable<QrResolveResponse> {
    const point = QR_LOCATIONS[code] ?? QR_LOCATIONS['river-entrance'];
    const session = sessions.get(sessionId ?? '') ?? this.createSession(point);
    session.currentPoint = point;
    session.status = 'ACTIVE';

    return of({
      sessionId: session.sessionId,
      hasActiveJourney: session.destinationNodeId !== null,
      venue: VENUE,
      startPoint: point,
      activeJourney: this.getJourney(session),
    });
  }

  getDestinations(venueId: number): Observable<Destination[]> {
    return venueId === VENUE.id
      ? of(DESTINATIONS)
      : throwError(() => new Error('Venue not found'));
  }

  calculateRoute(sessionId: string, destinationNodeId: number): Observable<RouteResponse> {
    const session = sessions.get(sessionId);
    const destination = DESTINATIONS.find((item) => item.nodeId === destinationNodeId);
    if (!session || !destination) return throwError(() => new Error('Route cannot be calculated'));

    session.destinationNodeId = destinationNodeId;
    return of(this.buildRoute(session.currentPoint, destination));
  }

  getCurrentJourney(sessionId: string): Observable<CurrentJourney> {
    const session = sessions.get(sessionId);
    const journey = session ? this.getJourney(session) : null;
    return journey ? of(journey) : throwError(() => new Error('Journey not found'));
  }

  updateLocation(sessionId: string, nodeId: number): Observable<RouteResponse> {
    const session = sessions.get(sessionId);
    const point = QR_LOCATIONS[nodeId];
    const destination = session?.destinationNodeId
      ? DESTINATIONS.find((item) => item.nodeId === session.destinationNodeId)
      : null;

    if (!session || !point || !destination) {
      return throwError(() => new Error('Active journey not found'));
    }

    session.currentPoint = point;
    return of(this.buildRoute(point, destination));
  }

  completeJourney(sessionId: string): Observable<CompleteJourneyResponse> {
    const session = sessions.get(sessionId);
    if (!session) return throwError(() => new Error('Session not found'));

    session.status = 'COMPLETED';
    session.destinationNodeId = null;
    return of({
      sessionId,
      status: 'COMPLETED',
      venueId: VENUE.id,
      venueName: VENUE.name,
      resumable: false,
    });
  }

  getFloorPlanMeta(floorId: number): Observable<FloorPlanMeta> {
    const plan = FLOOR_PLANS.find((item) => item.floorId === floorId);
    return plan ? of(plan) : throwError(() => new Error('Floor plan not found'));
  }

  getFloorPlanImageUrl(floorId: number): string {
    return FLOOR_PLAN_IMAGES[floorId] ?? FLOOR_PLAN_IMAGES[FLOORS.ground.id];
  }

  private createSession(currentPoint: StartPoint): MockSession {
    const session: MockSession = {
      sessionId: `library-session-${String(nextSessionId++).padStart(3, '0')}`,
      currentPoint,
      destinationNodeId: null,
      status: 'ACTIVE',
    };
    sessions.set(session.sessionId, session);
    return session;
  }

  private getJourney(session: MockSession): CurrentJourney | null {
    const destination = DESTINATIONS.find((item) => item.nodeId === session.destinationNodeId);
    return destination
      ? {
          sessionId: session.sessionId,
          status: session.status,
          venueId: VENUE.id,
          venueName: VENUE.name,
          currentPoint: session.currentPoint,
          destination: {
            nodeId: destination.nodeId,
            name: destination.name,
            floorId: destination.floorId,
            floorName: destination.floorName,
          },
          resumable: session.status === 'ACTIVE',
        }
      : null;
  }

  private buildRoute(from: StartPoint, destination: Destination): RouteResponse {
    const points: RoutePoint[] = [{ ...from, type: 'JUNCTION' }];

    if (from.floorId !== destination.floorId) {
      points.push({
        nodeId: ELEVATOR_POINTS[901].nodeId,
        name: ELEVATOR_POINTS[901].name,
        type: 'ELEVATOR',
        floorId: from.floorId,
        floorName: from.floorName,
        x: 700,
        y: 410,
      });
      points.push({
        ...ELEVATOR_POINTS[902],
        floorId: destination.floorId,
        floorName: destination.floorName,
      });
    }

    points.push(destination);
    return {
      venueId: VENUE.id,
      fromNodeId: from.nodeId,
      toNodeId: destination.nodeId,
      totalDistance: from.floorId === destination.floorId ? 260 : 480,
      estimatedTimeMinutes: from.floorId === destination.floorId ? 4 : 8,
      points,
    };
  }
}
